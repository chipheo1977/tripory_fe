```typescript
export type HttpMethod = 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE';
export type QueryValue = string | number | boolean | null | undefined;
export type HttpRequest = Readonly<{
  method: HttpMethod;
  path: string;                                   // nối sau baseUrl, ví dụ '/api/v1/itineraries'
  query?: Readonly<Record<string, QueryValue>>;   // null/undefined bị bỏ
  body?: unknown;                                 // FormData giữ nguyên; giá trị khác → JSON
  headers?: Readonly<Record<string, string>>;
  signal?: AbortSignal;
  timeoutMs?: number;
  sessionRefresh?: boolean;                       // false: login/register/refresh không vào nhánh refresh
}>;
export type HttpClient = Readonly<{ send(request: HttpRequest): Promise<unknown> }>;

// api-error.ts
export type ApiErrorKind = 'http' | 'network' | 'timeout' | 'aborted' | 'parse' | 'session_expired';
export type FieldErrors = Readonly<Record<string, readonly string[]>>;
export type ApiError = Readonly<{
  type: 'ApiError';
  kind: ApiErrorKind;
  status: number | null;
  code: string | null;
  message: string | null;
  fieldErrors?: FieldErrors;
}>;
export const SESSION_EXPIRED = 'session.expired';
export function makeApiError(fields: Omit<ApiError, 'type'>): ApiError;
export function isApiError(value: unknown): value is ApiError;
export function sessionExpiredError(): ApiError; // { kind: 'session_expired', status: 401, code: SESSION_EXPIRED, message: null }
```
