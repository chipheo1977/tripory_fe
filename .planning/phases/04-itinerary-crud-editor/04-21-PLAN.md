```tsx
import type { Metadata } from "next";
import { notFound } from "next/navigation";
import { parseItineraryId } from "@tripory/shared-kernel";
import { getViewer } from "@/composition/viewer.server";
import { ItineraryEditor } from "@/modules/itinerary/ui/itinerary-editor";

export const metadata: Metadata = { title: "Sửa lịch trình" };

// D-46: chỉ kiểm GUID và lấy viewer ở server; dữ liệu tải ở client qua BFF để Bearer stale được refresh
// trước khi kết luận 403 (Pitfall 11). Gate (editor)/layout đã bọc Suspense nên đọc params/cookies ở đây hợp lệ.
export default async function EditItineraryPage({ params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;
  const itineraryId = parseItineraryId(id);
  if (itineraryId === null) notFound();
  const viewer = await getViewer();
  // key theo viewer: Activity giữ state, đổi tài khoản không được thấy state cũ (Pitfall 4).
  return <ItineraryEditor key={viewer?.id ?? "guest"} itineraryId={itineraryId} viewerId={viewer?.id ?? null} />;
}
```
