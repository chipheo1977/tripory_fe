const legacyCovers = (legacy: string | null): string[] => {
  if (legacy === null) return [];
  return [legacy];

const coverList = (dto: Pick<ItineraryDetailDto, 'coverImageUrl' | 'coverImages'>): readonly string[] => {
  const source = dto.coverImages ?? legacyCovers(blankToNull(dto.coverImageUrl));
  const seen = new Set<string>(); const result: string[] = [];
  for (const raw of source) { const url = raw.trim(); if (url !== '' && !seen.has(url)) { seen.add(url); result.push(url); } }
  return Object.freeze(result);
};
const displayedCover = (dto: Pick<ItineraryDetailDto, 'coverImageUrl' | 'coverImages' | 'selectedCoverIndex'>): string | null =>
