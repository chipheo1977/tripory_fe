```tsx
function AddCoverPlaceholder({ className, onAdd }: Readonly<{ className: string; onAdd(): void }>) {
  /* giữ nguyên khối div "Thêm ảnh bìa" cũ: Icon ImagePlus + Button size="sm" variant="secondary" */
}
function TripCover({ src, fallbackSrc, className, onAdd }: Readonly<{ src: string | null; fallbackSrc: string | null; className: string; onAdd(): void }>) {
  const hasAnyCover = hasCover(src) || hasCover(fallbackSrc); // một cấp `||`, không `? :`
  if (hasAnyCover) return <CoverImage className={className} fallbackSrc={fallbackSrc} src={src} />;
  return <AddCoverPlaceholder className={className} onAdd={onAdd} />;
}
```
Khối "Thêm ảnh bìa" (CTA) chỉ hiện khi không có cả ảnh đang hiển thị lẫn ảnh mặc định (hành vi cũ cho lịch trình không có điểm đến).
