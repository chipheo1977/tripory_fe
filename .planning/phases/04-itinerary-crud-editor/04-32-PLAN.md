```ts
import type { CalendarDate } from './calendar-date';
import type { Itinerary } from './itinerary';

/** Bốn trường người dùng sửa trong "Thông tin chuyến đi" + chỉ số ảnh đang hiển thị (D-99). */
export type ItineraryMetadata = Readonly<{
  title: string;
  description: string | null;
  startDate: CalendarDate | null;
  coverImages: readonly string[];
  /** Vị trí ảnh hiển thị trong `coverImages`; `null` khi danh sách rỗng. */
  selectedCoverIndex: number | null;
}>;

export const metadataOf = (itinerary: Itinerary): ItineraryMetadata => ...;
export const withMetadata = (itinerary: Itinerary, input: ItineraryMetadata): Itinerary => ...;
export const isSameMetadata = (a: ItineraryMetadata, b: ItineraryMetadata): boolean => ...;
```
- `metadataOf`: `selectedCoverIndex` = `null` khi `coverImages.length === 0`; ngược lại `Math.max(coverImages.indexOf(itinerary.coverImageUrl ?? ''), 0)` (ảnh hiển thị không có trong danh sách thì coi như ảnh đầu, giữ bất biến "danh sách không rỗng thì có ảnh được chọn"); các trường còn lại lấy thẳng.
- `withMetadata`: trả bản sao `Object.freeze({ ...itinerary, title, description, startDate, coverImages: Object.freeze([...input.coverImages]), coverImageUrl })` với `coverImageUrl = input.selectedCoverIndex === null ? null : (input.coverImages[input.selectedCoverIndex] ?? null)`. Không đụng `destinationCode`, `interests`, `days`, `waypoints`, `isPublic`, `totalDistanceKm`.
- `isSameMetadata`: so `title`, `description`, `startDate`, `selectedCoverIndex` bằng `===` và `coverImages` từng phần tử (cùng độ dài).
`metadata.test.ts` (dùng fixture từ `test-fixtures.ts`): `metadataOf` ảnh rỗng → `selectedCoverIndex` null; hai ảnh với `coverImageUrl` là ảnh thứ hai → `1`; `coverImageUrl` không thuộc danh sách (hoặc null) với danh sách không rỗng → `0`; `withMetadata` đổi tiêu đề/mô tả/ngày/ảnh và `coverImageUrl` theo chỉ số, `selectedCoverIndex` null → `coverImageUrl` null, chỉ số ngoài khoảng → `coverImageUrl` null; giữ nguyên `waypoints`, `days`, `destinationCode`, `interests` (cùng tham chiếu); không sửa input; `isSameMetadata` true/false cho từng trường, mảng ảnh khác thứ tự → false.
`domain/index.ts`: thêm `export * from './metadata';`. `exports.test.ts`: thêm `'metadataOf'`, `'withMetadata'`, `'isSameMetadata'` vào `DOMAIN_NAMES`.
