```ts
export const TEXT_AUTOSAVE_DEBOUNCE_MS = 800;
export type AutosaveFailure = Readonly<{ alert: string | null; fieldErrors: Readonly<Record<string, string>> }>;
export type AutosaveState = Readonly<{ timerActive: boolean; queued: boolean; failure: AutosaveFailure | null; dirty: boolean }>;
export type AutosaveOutcome = Readonly<{ ok: true }> | Readonly<{ ok: false; failure: AutosaveFailure }>;
export type AutosaveDeps = Readonly<{
  send(payload: MetadataInput): void;                                   // = mutation.mutate
  onBackgroundFailure(failure: AutosaveFailure, retry: () => void): void; // lỗi khi form đã đóng
  debounceMs?: number;
  setTimer?(callback: () => void, ms: number): unknown;
  clearTimer?(handle: unknown): void;
}>;
export type MetadataAutosave = Readonly<{
  attach(provider: () => AutosaveSnapshot): void;
  detach(): void;
  openForm(baseline: MetadataInput): void;
  closeForm(): void;
  schedule(): void;   // ô chữ đổi: khởi động lại đồng hồ 800 ms
  flush(): void;      // blur/đóng/unmount: hủy đồng hồ và request() ngay
  request(): void;    // ngày khởi hành, thao tác ảnh: gửi ngay
  settled(outcome: AutosaveOutcome): void; // gọi từ callback cấp hook của mutation
  retry(): void;
  getState(): AutosaveState;
  subscribe(listener: () => void): () => void;
}>;
export function createMetadataAutosave(deps: AutosaveDeps): MetadataAutosave;
```
Trạng thái nội bộ: `provider`, `lastSaved: MetadataInput | null` (payload đã được BE xác nhận hoặc mốc ban đầu), `lastSent: MetadataInput | null`, `inFlight: MetadataInput | null`, `queued: MetadataInput | null`, `lastBuilt: MetadataInput | null`, `formOpen: boolean`, `timer`, `failure`. Hàm:
- `request()`: nếu `provider === null` hoặc `lastSaved === null` thì thoát. `payload = buildAutosavePayload({ lastSaved, snapshot: provider() })` (getter đọc LÚC CHẠY); `lastBuilt = payload`; so với `queued ?? lastSent`: nếu `lastSent !== null && isSameMetadata(payload, queued ?? lastSent)` thì thoát; nếu `inFlight !== null` thì `queued = payload` (gộp, lệnh cuối thắng); ngược lại `sendNow(payload)`. Gọi `notify()` (thông báo subscriber) khi `queued` đổi.
- `sendNow(p)`: `inFlight = p; lastSent = p; deps.send(p)`.
- `schedule()`: hủy đồng hồ cũ; đặt đồng hồ mới `setTimer(() => { timer = null; request(); notify() }, debounceMs ?? 800)`; `notify()`.
- `flush()`: hủy đồng hồ (nếu có) rồi `request()`; `notify()`.
- `settled(outcome)`: `const sent = inFlight; inFlight = null;` Nếu `ok`: `lastSaved = sent ?? lastSaved; failure = null;` nếu `queued !== null` thì `const next = queued; queued = null; sendNow(next)`. Nếu không `ok`: `failure = outcome.failure; lastSent = lastSaved; queued = null;` rồi nếu `!formOpen && failure.alert !== null` thì `deps.onBackgroundFailure(failure, retry); failure = null`. `notify()`.
- `retry()`: `failure = null`; `payload = provider !== null && lastSaved !== null ? buildAutosavePayload(...) : lastBuilt`; nếu `payload === null` thoát; `lastBuilt = payload`; nếu `inFlight !== null` thì `queued = payload`, ngược lại `sendNow(payload)` (gửi dù trùng `lastSent`, vì `lastSent` đã đặt lại về `lastSaved` khi lỗi); `notify()`.
- `openForm(baseline)`: `formOpen = true; failure = null`; chỉ khi `inFlight === null && queued === null` mới đặt `lastSaved = lastSent = baseline`.
- `closeForm()`: `flush()`; `formOpen = false`; nếu `failure !== null && failure.alert !== null` thì `deps.onBackgroundFailure(failure, retry)`; `failure = null`; `notify()`.
- `attach(provider)`/`detach()`: gán/xóa `provider` (sau `detach` bộ điều khiển vẫn gửi nốt `queued`).
- `getState()`: `{ timerActive: timer !== null, queued: queued !== null, failure, dirty: timerActive || queued || (failure?.alert ?? null) !== null }`, trả CÙNG tham chiếu cho tới khi `notify()` (cho `useSyncExternalStore`). `subscribe` trả hàm hủy. Không có `dispose` (hook chỉ `flush()` khi dọn, xem T3).
