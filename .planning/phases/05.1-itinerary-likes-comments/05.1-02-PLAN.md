$base = "<BE_URL>/api/v1/itineraries/<ID>"

Invoke-RestMethod -Method Put -Uri "$base/like" -Headers $b

$c = Invoke-RestMethod -Method Post -Uri "$base/comments" -Headers $b -Body $body
$c.data

Invoke-RestMethod -Method Delete -Uri "$base/comments/$($c.data.id)" -Headers $b
